A bundle is a kit whose `adds` make several assemblies, each on machines taken from the bundle's own roles.

| Bundle | Roles | Adds |
|---|---|---|
| `base` | `default` with `device`, `layout` (a ref), `authorizedKeys`, `hardware` | disko, facter, emergency-access, and sshd as server, on the same machines. No users: a user needs a name per machine |
| `homelab` | `server` (exactly 1), `node`; the settings of `base` and `prefix` | base on both roles; wireguard (server controller, nodes peers); borg (server backs up); monitoring (server scrapes); dns on the server |

The engine adds one level, so `homelab` also lists what `base` and `borg` add. A role's `settings` may be a
function of the machine, for a device per machine. You import the kits a bundle uses, then write one assembly:


```nix title="modules/assemblies.nix"
# A bundle is a kit whose assemblies make more assemblies: one `homelab` assembly names the server and the nodes,
# and the base, wireguard, borg, monitoring and dns assemblies follow from those two roles. Import the kits it uses.
{ yen, inputs, ... }:
let
  inherit (yen) host kit;
  k = inputs.yen.lib.kits;
  shared = import ../shared.nix yen;
in
{
  imports = [
    k.homelab
    k.base
    k.disko
    k.facter
    k.sshd
    k.emergency-access
    (k.wireguard { root = ../vars; })
    k.borg
    k.monitoring
    k.dns
  ];
  yen.host = {
    lb.addr = "10.0.0.1";
    web.addr = "10.0.0.2";
    nas.addr = "10.0.0.3";
  };
  yen.assembly.lab = {
    kit = kit.homelab;
    roles = {
      server = {
        at = host.nas;
        settings = shared // {
          device = "/dev/sda /dev/sdb";
          layout = yen.layout.zfs-mirror;
        };
      };
      node = {
        at = [
          host.lb
          host.web
        ];
        settings = shared;
      };
    };
  };
}
```

```nix title="modules/fleet.nix"
# The fleet's words: hosts with an address, members of assemblies, and the sinks the library kits write to.
{ yen, lib, ... }:
let
  kits = yen.lib.batteries.kits;
  inherit (yen.lib) opt holds;
  attrs = yen.lib.is lib.isAttrs;
  # a kit that adds assemblies of another kit (borg adds sshd), per assembly of its own
  added =
    a:
    lib.mapAttrsToList (
      name: v:
      v
      // {
        kind = "assembly";
        inherit name;
      }
    ) (a.kit.adds a);
  own = lib.attrValues yen.assembly;
  assemblies = own ++ lib.concatMap added own;
  machineOf = s: if holds yen.kinds.member s then s.parent else s;
in
{
  yen.kinds.fleet = {
    spawns.host = _: yen.host;
    sinks = {
      out.reduce = yen.lib.reduce.merge;
      exports.reduce = kits.exports { isMember = holds yen.kinds.member; };
      generators = {
        reduce = yen.lib.batteries.vars.generators { inherit machineOf; };
        out = yen.lib.batteries.vars.vars-plan {
          from = yen.kinds.fleet.sinks.generators;
          machines = lib.attrNames yen.host;
          recipientsOf = _: [ "age1admin" ];
        };
      };
    };
  };
  yen.kinds.host = {
    fields.addr = lib.types.str;
    spawns.member = kits.members { inherit assemblies holds; };
    # the NixOS keys of a host, as data
    sinks.disks.reduce = yen.lib.reduce.merge; # kit disko gives disks here
    sinks.nixos = {
      reduce = yen.lib.reduce.merge;
      out =
        { scope, sinks, ... }:
        {
          config.${scope.name} = sinks.nixos;
        };
    };
  };
  yen.kinds.member = { };
  yen.kinds.kit.fields = {
    roles = opt attrs { };
    constraints = opt attrs { };
    exports = opt (yen.lib.is lib.isList) [ ];
    adds = opt (yen.lib.is lib.isFunction) (_: { });
  };
  yen.kinds.assembly.fields = {
    kit = yen.kinds.kit;
    roles = opt attrs { };
  };
  yen.aspects.contract = {
    at = yen.kinds.fleet;
    errors = kits.contract {
      inherit assemblies holds;
      machines = lib.attrValues yen.host;
    };
  };
  yen.aspects.settings-typed = {
    at = yen.kinds.member;
    errors = kits.settings-typed;
  };
  # a generator that an export carries goes to the machines of its `deploy` predicate
  yen.aspects.generators-from-exports = {
    at = yen.kinds.host;
    generators =
      { scope, exports, ... }:
      let
        gens = lib.concatMap (
          x: lib.filter (g: holds g.deploy scope) (lib.attrValues (x.value.generators or { }))
        ) exports;
      in
      if gens == [ ] then
        null
      else
        {
          name = "from-exports";
          files = lib.foldl' (a: g: a // g.files) { } gens;
        };
  };
  yen.fleet.main = { };
  yen.flake = { out, ... }: out;
}
```

## The contract

`homelab` allows exactly one server. Two:


```nix title="variants/two-servers.nix"
# a homelab with two servers: exactly one is allowed
{ yen, ... }:
{
  yen.assembly.lab2 = {
    kit = yen.kit.homelab;
    roles = {
      server = {
        at = [
          yen.host.nas
          yen.host.lb
        ];
        settings = import ../shared.nix yen;
      };
      node = {
        at = yen.host.web;
        settings = import ../shared.nix yen;
      };
    };
  };
}
```

```nix title="result"
[
  "merge conflict at services.prometheus.scrapeConfigs"

  "wireguard: nas has no wg-lab2-wg.pub public key"

]
```

And none:


```nix title="variants/no-server.nix"
# a homelab with nodes and no machine for the server role
{ yen, ... }:
{
  yen.assembly.lab0 = {
    kit = yen.kit.homelab;
    roles = {
      server = {
        at = _: false;
        settings = import ../shared.nix yen;
      };
      node = {
        at = yen.host.web;
        settings = import ../shared.nix yen;
      };
    };
  };
}
```

```nix title="result"
]
```
