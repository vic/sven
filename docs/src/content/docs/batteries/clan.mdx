[clan](https://clan.lol) describes services as data: a service has roles, an instance places machines in roles by
name or by tag, each role has typed settings, members export values to each other, and services declare the
secrets they generate. `yen.lib.batteries.clan` gives the same rules. You keep the words: here the machine kind
is `machine` and the member kind is `member`.

| Battery | Value of | Does |
|---|---|---|
| `members { instances; services; labelsOf; }` | `spawns.<member kind>` of a machine kind | one member per (instance, role), settings layered: interface default, role, machine |
| `exports { services; isMember; }` | `reduce` of the exports sink | members write their own key; another scope overrides by key; an unlisted export is an error |
| `contract { instances; services; machines; labelsOf; }` | `errors` | unknown service or role, no roles, min and max machines per role, `maxInstances` |
| `settings-typed { services; }` | `errors` at a member | every interface setting is set and has its type |
| `unknown-tags { instances; machines; labelsOf; }` | a warnings sink | tags that match no machine |
| `extra-modules` | a class key at a member | the role's `extraModules` |
| `generators { machineOf; }` | `reduce` of the generators sink | shared generators once, the rest per machine; a missing dependency is an error |
| `vars-plan { from; machines; recipientsOf; }` | `out` of the generators sink | per machine: secret files with their paths, recipients, generation order |
| `secret-reads { from; reads; }` | `errors` | an eval-time read of a secret file |
| `public-value { root; }` | a function | a public var read at eval time, with a default |


```nix title="modules/inventory.nix"
# A clan-like inventory: services and instances are data; machines take roles by name or tag.
{ yen, lib, ... }:
let
  clan = yen.lib.batteries.clan;
  str = lib.types.str;
  machines = lib.attrValues yen.machine;
  labelsOf = m: [ "all" ] ++ m.tags;
  # what a service is: its roles, a typed interface per role, limits, the exports it may give
  services = {
    borg = {
      roles.server = { };
      roles.client.interface.dir = {
        type = str;
        default = "/var/backup";
      };
      constraints.roles.server.max = 1;
      exports = [ "endpoint" ];
    };
  };
  instances.daily = {
    module = "borg";
    roles.server.tags = [ "storage" ];
    roles.client = {
      tags = [
        "all"
        "laptops"
      ]; # no machine has `laptops`: a warning
      extraModules = [ { services.borg.verbose = true; } ];
      machines.web.settings.dir = "/srv/backup";
    };
  };
in
{
  yen.kinds.fleet = {
    spawns.machine = _: yen.machine;
    sinks.out.reduce = yen.lib.reduce.merge;
  };
  yen.kinds.machine = {
    fields.tags = yen.lib.opt (yen.lib.is lib.isList) [ ];
    # one member scope per (instance, role) the machine takes part in
    spawns.member = clan.members { inherit instances services labelsOf; };
    sinks.config = {
      reduce = yen.lib.reduce.merge;
      out =
        { scope, sinks, ... }:
        {
          config.${scope.name} = sinks.config;
        };
    };
  };
  yen.kinds.member = { };
  yen.kinds.fleet.sinks.exports = {
    reduce = clan.exports {
      inherit services;
      isMember = yen.lib.holds yen.kinds.member;
    };
    out =
      { sinks, ... }:
      {
        exports = sinks.exports;
      };
  };
  yen.fleet.main = { };
  yen.machine = {
    nas.tags = [ "storage" ];
    web = { };
  };
  yen.flake = { out, ... }: out;

  yen.kinds.fleet.sinks.warnings.out =
    { sinks, ... }:
    {
      warnings = sinks.warnings;
    };
  yen.aspects = {
    unknown-tags = {
      at = yen.kinds.fleet;
      warnings = clan.unknown-tags { inherit instances machines labelsOf; };
    };
    extra-modules = {
      at = yen.lib.all [
        yen.kinds.member
        (s: s.extraModules != [ ])
      ];
      config = clan.extra-modules;
    };
    contract = {
      at = yen.kinds.fleet;
      errors = clan.contract {
        inherit
          instances
          services
          machines
          labelsOf
          ;
      };
    };
    settings-typed = {
      at = yen.kinds.member;
      errors = clan.settings-typed { inherit services; };
    };
    # perInstance: once per member. The server exports its endpoint; clients read it back.
    borg-server = {
      at = yen.lib.all [
        yen.kinds.member
        (s: s.service == "borg" && s.role == "server")
      ];
      exports =
        { scope, ... }:
        {
          endpoint = "${scope.parent.name}:22";
        };
    };
    borg-client = {
      at = yen.lib.all [
        yen.kinds.member
        (s: s.service == "borg" && s.role == "client")
      ];
      config =
        { scope, exports, ... }:
        {
          backup.${scope.instance} = {
            inherit (scope.settings) dir;
            to = map (x: x.value.endpoint) (lib.filter (x: x.role == "server") exports);
          };
        };
    };
  };
}
```

Each client reads the server's endpoint from the exports, with its own settings layer:

## Vars


```nix title="modules/vars.nix"
# clan vars: generators declared by services, a plan per machine, public values read at eval time.
{ yen, lib, ... }:
let
  clan = yen.lib.batteries.clan;
  machineOf = s: if yen.lib.holds yen.kinds.member s then s.parent else s;
in
{
  yen.kinds.fleet.sinks = {
    generators = {
      reduce = clan.generators { inherit machineOf; };
      out = clan.vars-plan {
        from = yen.kinds.fleet.sinks.generators;
        machines = lib.attrNames yen.machine;
        recipientsOf = m: [ "age1admin" ];
      };
    };
    public-reads.reduce = yen.lib.reduce.entries;
  };
  yen.aspects.ssh-vars = {
    at = yen.kinds.machine;
    # a function of the scope: one per machine (static content goes once per receiver)
    generators =
      { scope, ... }:
      {
        name = "ssh";
        files.key.secret = true;
        files."key.pub".secret = false;
      };
    config =
      { scope, ... }:
      {
        sshPub = clan.public-value { root = ../vars/per-machine; } scope.name "ssh" "key.pub" "absent";
      };
  };
  yen.aspects.borg-vars = {
    at = yen.lib.all [
      yen.kinds.member
      (s: s.service == "borg" && s.role == "client")
    ];
    generators =
      { scope, ... }:
      {
        name = "borg-${scope.instance}";
        dependencies = [ "ssh" ];
        files.passphrase.neededFor = "users";
      };
  };
  yen.aspects.secret-reads = {
    at = yen.kinds.fleet;
    errors = clan.secret-reads {
      from = yen.kinds.fleet.sinks.generators;
      reads = yen.kinds.fleet.sinks.public-reads;
    };
  };
}
```

```nix title="result"
  }

    file = "key";
    generator = "ssh";
    neededForUsers = false;
    path = "/run/secrets/vars/ssh/key";
  }
]
```

## The contract

borg allows one server. A second machine with the `storage` tag makes two:


```nix title="result"
[
  "clan: instance daily role server: 2 machines, at most 1"
]
```
