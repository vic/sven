`yen.lib.batteries.nixos` holds the NixOS rules that most fleets write. NixOS is the domain of these batteries:
they set NixOS options. They still name no yen kind, sink or aspect. You write `yen.kinds.host`, the `nixos` and
`unfree` sinks and every aspect, in your own words.

| Battery | Value of | Den |
|---|---|---|
| `system { from; nixpkgs; }` | the `out` key of the sink that collects NixOS modules | the `den.hosts` schema |
| `hostname` | a NixOS key of an aspect | `den.batteries.hostname` |
| `define-user` | a NixOS key of an aspect | `den.batteries.define-user` |
| `primary-user` | a NixOS key of an aspect | `den.batteries.primary-user` |
| `unfree { from; }` | the NixOS key of a sink of package names | `den.batteries.unfree` |
| `insecure { from; }` | the NixOS key of a sink of package names | `den.batteries.insecure` |
| `user-shell { shell; }` | a NixOS key; `shell` reads the scope | `den.batteries.user-shell` |
| `autologin { user; }` | a NixOS key; `user` reads a ref or a name | `den.batteries.tty-autologin` |
| `firewall { from; }` | the NixOS key of a sink of ports (`22` or `{ tcp; udp; }`) | quirks `firewall` |
| `hosts { from; }` | a NixOS key; /etc/hosts and known_hosts of the other machines (`mesh`) | sini hostsfile |
| `remote-builders { from; systemOf; }` | a NixOS key; `nix.buildMachines` of the other machines (`mesh`) | sini remote-build-client |
| `persist { from; root; hideMounts; }` | the NixOS key of a sink of paths (impermanence) | quirks `persist` |
| `facter { report; module; }` | a NixOS key, at hosts that have a report | `den.aspects.facter` |


```nix title="modules/fleet.nix"
# Your words: hosts build NixOS and hold users. The batteries fill values; every key here is yours.
{
  yen,
  lib,
  inputs,
  ...
}:
let
  nixos = yen.lib.batteries.nixos;
in
{
  yen.kinds.fleet = {
    spawns.host = _: yen.host;
    sinks.out.reduce = yen.lib.reduce.merge;
  };
  yen.kinds.host = {
    spawns.user = s: lib.filterAttrs (_: u: lib.any (yen.lib.same s) u.hosts) yen.user;
    sinks.nixos.out = nixos.system {
      from = yen.kinds.host.sinks.nixos;
      inherit (inputs) nixpkgs;
    };
    # package names from any feature or user; one module per host allows them
    sinks.unfree.nixos = nixos.unfree { from = yen.kinds.host.sinks.unfree; };
    sinks.insecure.nixos = nixos.insecure { from = yen.kinds.host.sinks.insecure; };
  };
  yen.kinds.user.fields = {
    hosts = yen.lib.is lib.isList;
    shell = yen.lib.opt (lib.types.enum [
      "bash"
      "fish"
      "zsh"
    ]) "bash";
  };
  yen.kinds.host.fields.autologin = yen.lib.maybe yen.kinds.user;

  yen.fleet.main = { };
  yen.flake = { out, ... }: out;
}
```

```nix title="modules/aspects.nix"
# Den's hostname, define-user and primary-user, as values of aspects you name.
{ yen, ... }:
let
  nixos = yen.lib.batteries.nixos;
in
{
  yen.aspects.hostname = {
    at = yen.kinds.host;
    nixos = nixos.hostname;
  };
  yen.aspects.accounts = {
    at = yen.kinds.user; # a user has no `nixos` sink: its account goes into each of its hosts
    nixos = nixos.define-user;
  };
  yen.aspects.admin = {
    at = yen.user.ana;
    nixos = nixos.primary-user;
  };
  # Den's user-shell and tty-autologin: the battery reads the field you name
  yen.aspects.shells = {
    at = yen.kinds.user;
    nixos = nixos.user-shell { shell = u: u.shell; };
  };
  yen.aspects.autologin = {
    at = yen.kinds.host;
    nixos = nixos.autologin { user = h: h.autologin; };
  };
}
```

```nix title="modules/network.nix"
# Ports, peers and builders: sinks you name, NixOS values from batteries.
{ yen, lib, ... }:
let
  nixos = yen.lib.batteries.nixos;
  str = lib.types.str;
in
{
  yen.kinds.host = {
    fields = {
      addr = str;
      sshKey = str;
    };
    sinks.ports.nixos = nixos.firewall { from = yen.kinds.host.sinks.ports; };
  };
  # every host gives a peer record and a builder record to the fleet; each host reads the others'
  yen.kinds.fleet.sinks = {
    peers.reduce = yen.lib.reduce.entries;
    builders.reduce = yen.lib.reduce.entries;
  };
  yen.aspects.peer = {
    at = yen.kinds.host;
    peers =
      { host, ... }:
      {
        inherit (host) name addr;
        key = host.sshKey;
      };
    builders =
      { host, ... }:
      {
        hostName = host.addr;
        system = "x86_64-linux";
        maxJobs = 4;
      };
    nixos = nixos.hosts { from = yen.kinds.fleet.sinks.peers; };
  };
  yen.aspects.builds = {
    at = yen.kinds.host;
    nixos = nixos.remote-builders { from = yen.kinds.fleet.sinks.builders; };
  };
  yen.aspects.ssh = {
    at = yen.kinds.host;
    ports = 22;
    nixos.services.openssh.enable = true;
  };
  yen.aspects.dns = {
    at = yen.host.nas;
    ports = {
      tcp = [ 53 ];
      udp = [ 53 ];
    };
  };
}
```

```nix title="modules/storage.nix"
# Paths that survive a reboot, from any feature (impermanence), and hardware from a facter report where one exists.
{ yen, inputs, ... }:
let
  nixos = yen.lib.batteries.nixos;
in
{
  yen.kinds.host.sinks.persist.nixos = nixos.persist {
    from = yen.kinds.host.sinks.persist;
    hideMounts = true;
  };
  yen.aspects.impermanence = {
    at = yen.kinds.host;
    nixos.imports = [ inputs.impermanence.nixosModules.impermanence ];
  };
  yen.aspects.ssh-state = {
    at = yen.kinds.host;
    persist.files = [ "/etc/ssh/ssh_host_ed25519_key" ];
  };
  yen.aspects.ana-state = {
    at = yen.user.ana; # from a user: up to each of her hosts
    persist = "/home/ana/.local/share";
  };
  # facter: only hosts with a report under modules/<host>/
  yen.aspects.facter = {
    at = s: builtins.pathExists (./. + "/${s.name}/facter.json");
    nixos = nixos.facter {
      report = h: ./. + "/${h.name}/facter.json";
      module = inputs.nixos-facter-modules.nixosModules.facter;
    };
  };
}
```

Each battery is a few lines. The pages [Every Den battery](/den/batteries/) and
[Unfree packages from many files](/nixos/unfree/) write the same rules by hand. When a battery does not fit your
fleet, copy those lines and change them: see [Write your own](/batteries/generic/#write-your-own).
