Some questions are rules: "a host serves when it runs a web server and is not drained; a host is drained when a host
it depends on is down." `yen.lib.datalog` solves ground rules with negation. Each rule is `{ head; pos; neg; }`.
You ground the rules with Nix: one set per host. Each atom is a relation of `yen.relations` called by reference,
and each term an instance: no relation or host is written as a string. In an answer, an instance is `{ kind; name; }`.


```nix title="modules/fleet.nix"
{ yen, lib, ... }:
let
  inherit (yen.lib) opt is;
in
{
  yen.kinds.fleet = {
    spawns.host = _: yen.host;
    sinks.out.reduce = yen.lib.reduce.merge;
  };
  yen.kinds.host.fields = {
    web = opt lib.types.bool false;
    maintenance = opt lib.types.bool false;
    dependsOn = opt (is lib.isList) [ ];
  };

  yen.fleet.main = { };
  yen.host.db-1 = { };
  yen.host.db-2.maintenance = true;
  yen.host.web-1 = {
    web = true;
    dependsOn = [ yen.host.db-1 ];
  };
  yen.host.web-2 = {
    web = true;
    dependsOn = [ yen.host.db-2 ];
  };
  yen.flake = { out, ... }: out;
}
```

```nix title="modules/policy.nix" {17-29}
# Ground rules { head; pos; neg; }, one set per host. Each atom is a relation called by reference, each term an instance.
{ yen, lib, ... }:
let
  D = yen.lib.datalog;
  inherit (yen.relations)
    web
    up
    serves
    drained
    primary
    ;
  all.instances = yen.kinds.host; # a tree request: every host, with its data
  rules =
    hosts:
    lib.concatMap (
      h:
      lib.optional h.web { head = web h; }
      ++ lib.optional (!h.maintenance) { head = up h; }
      ++ [
        {
          head = serves h;
          pos = [ (web h) ];
          neg = [ (drained h) ];
        }
      ]
      ++ map (d: {
        head = drained h;
        neg = [ (up d) ];
      }) h.dependsOn
    ) hosts
    # a choice: exactly one primary database, and the policy does not say which
    ++ [
      {
        head = primary yen.host.db-1;
        neg = [ (primary yen.host.db-2) ];
      }
      {
        head = primary yen.host.db-2;
        neg = [ (primary yen.host.db-1) ];
      }
    ];
  dbs = [
    yen.host.db-1
    yen.host.db-2
  ];
in
{
  yen.relations = {
    web.arity = 1;
    up.arity = 1;
    serves.arity = 1;
    drained.arity = 1;
    primary.arity = 1;
  };
  yen.aspects.serving = {
    at = yen.kinds.fleet;
    cycle =
      { tree, ... }:
      let
        hosts = lib.head (tree.select all);
        rs = rules hosts;
        model = D.wfs rs;
        names = map (h: h.name);
      in
      {
        tree = [ all ];
        out = [
          {
            serving = {
              serves = names (lib.filter (h: model.holds (serves h)) hosts);
              undefined = model.undefined; # well-founded: the choice stays open
              # stable models: each choice
              choices = map (m: names (lib.filter (d: lib.elem (D.atom (primary d)) m) dbs)) (D.stable rs);
              whyNot = D.whyNot rs model (serves yen.host.web-2);
              proof = D.proof rs model (serves yen.host.web-1);
            };
          }
        ];
      };
  };
}
```

| Function | Answer |
|---|---|
| `wfs rules` | the well-founded model: `{ true; false; undefined; }`. No strata needed. |
| `stable rules` | every stable model: one per consistent choice |
| `why rules model atom` | the rules that prove an atom |
| `proof rules model atom` | a proof tree; a positive cycle shows `cycle = true` |
| `whyNot rules model atom` | per rule: the `missing` positive atoms and the `blocking` negative ones |
| `judge rules a notA` | `supported`, `refuted`, `both`, `neither` or `unknown` |
| `facts atoms` | atoms as rules with no body |

`web-2` does not serve: its database `db-2` is in maintenance, so `web-2` is drained. `whyNot` says so.
The primary database is undefined in the well-founded model. The stable models list the two choices.


:::note[Undefined is an answer]
An atom the rules cannot decide is `undefined`, never a guess. When that matters, give an error record for it.
[Invariants](/query/invariants/) use the same solver, and treat an undefined violation as `incomplete`, never a pass.
:::
