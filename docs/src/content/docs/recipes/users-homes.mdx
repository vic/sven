A person uses several machines. Some run NixOS, some only home-manager. In yen the person is data,
and their home is one aspect that is active in both places.


```nix title="modules/fleet.nix" {17-20,22-28}
# Hosts build NixOS and hold their users. Homes are standalone home-manager configurations.
{
  yen,
  lib,
  inputs,
  ...
}:
let
  inherit (yen.lib) same is;
in
{
  yen.kinds.fleet = {
    spawns = {
      host = _: yen.host;
      home = _: yen.home;
    };
    sinks.out.reduce = yen.lib.reduce.merge;
  };
  yen.kinds.host = {
    spawns.user = s: lib.filterAttrs (_: u: lib.any (same s) u.hosts) yen.user;
    sinks.nixos.out =
      { host, nixos, ... }:
      {
        nixosConfigurations.${host.name} = inputs.nixpkgs.lib.nixosSystem { modules = nixos; };
      };
  };
  yen.kinds.user = {
    fields.hosts = is lib.isList;
    sinks.homeManager.nixos =
      { user, homeManager, ... }:
      {
        home-manager.users.${user.name}.imports = homeManager;
      };
  };
  yen.kinds.home = {
    fields = {
      user = yen.kinds.user;
      system = lib.types.str;
    };
    sinks.homeManager.out =
      { home, homeManager, ... }:
      {
        homeConfigurations.${home.name} = inputs.home-manager.lib.homeManagerConfiguration {
          pkgs = inputs.nixpkgs.legacyPackages.${home.system};
          modules = homeManager;
        };
      };
  };
  yen.aspects.home-manager = {
    at = yen.lib.has yen.kinds.user;
    nixos = {
      imports = [ inputs.home-manager.nixosModules.home-manager ];
      home-manager.useGlobalPkgs = true;
    };
  };

  yen.fleet.main = { };
  yen.flake = { out, ... }: out;
}
```

- A `user` lives under each host in its `hosts`. Its `homeManager` sink forwards into the host's NixOS.
- A `home` lives under the fleet. Its `homeManager` sink builds `homeConfigurations.<name>`.


```nix title="modules/alice.nix" {6}
# alice: an account on the server, and a standalone home for her laptop (any Linux, no NixOS).
{ yen, ... }:
let
  alice = yen.user.alice;
  # her own scopes, and every home that belongs to her
  hers = [
    alice
    (s: s.kind == "home" && yen.lib.same s.user alice)
  ];
in
{
  yen.user.alice.hosts = [ yen.host.server ];
  yen.home."alice@laptop" = {
    user = alice;
    system = "x86_64-linux";
  };

  yen.aspects.alice-account = {
    at = alice;
    nixos.users.users.alice = {
      isNormalUser = true;
      extraGroups = [ "wheel" ];
    };
  };
  yen.aspects.alice-home = {
    at = hers; # one home config, in NixOS and standalone
    homeManager = {
      programs.git = {
        enable = true;
        userName = "Alice";
      };
      home.stateVersion = "25.05";
    };
  };
  yen.aspects.standalone = {
    at = yen.kinds.home;
    homeManager =
      { home, ... }:
      {
        home.username = home.user.name;
        home.homeDirectory = "/home/${home.user.name}";
      };
  };
}
```

:::tip[A home that reads its host]
A home can also live under a host: give `host` a `spawns.home`. The home then reads the host's sinks like any child,
and still builds its own `homeConfigurations`. A home's sink with no keys keeps its content away from the host.
See [Routing and walls](/concepts/routing/#walls).
:::
