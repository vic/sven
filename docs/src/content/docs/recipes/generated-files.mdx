Some repo files are views of your fleet: the flake's inputs, a README with the host list, a CI matrix.
Write them from the data, and a check fails when a committed file is stale.
[sini/files](https://github.com/sini/files) does the writing. sven gives it the content.

These docs are made this way: every page you read is a file that the docs fleet writes. So is sven's CI workflow:
each stage is an aspect, and one job per system comes from the reference fleet's `ci.json`
([docs/modules/ci.nix](https://github.com/vic/sven/blob/main/docs/modules/ci.nix)).


```nix title="modules/fleet.nix" {11,12-20}
# The fleet writes its own repo files. `files` collects github:sini/files modules; `write-files` writes them.
{
  sven,
  lib,
  inputs,
  ...
}:
let
  pkgs = inputs.nixpkgs.legacyPackages.x86_64-linux;
in
{
  sven.kinds.fleet = {
    spawns.host = _: sven.host;
    sinks = {
      out.reduce = sven.lib.reduce.merge;
      inputs.reduce = sven.lib.reduce.merge; # flake inputs, declared by the aspects that use them
      files.out =
        { files, ... }:
        let
          e = lib.evalModules {
            modules = [
              inputs.files.module
              {
                _module.args.pkgs = pkgs;
                files.root = inputs.self;
                files.relativeRoot = "docs/examples/generated-files";
              }
            ]
            ++ files;
          };
        in
        {
          packages.x86_64-linux.write-files = e.config.files.writer.drv;
          checks.x86_64-linux = e.config.files.checks; # each committed file must match
        };
    };
  };
  sven.kinds.host.sinks.nixos.out =
    { host, nixos, ... }:
    {
      nixosConfigurations.${host.name} = inputs.nixpkgs.lib.nixosSystem { modules = nixos; };
    };

  sven.fleet.main = { };
  sven.host.web = { };
  sven.flake = { out, ... }: out;
}
```

## Inputs live with their users

Each aspect declares the flake inputs it uses on the `inputs` sink. The host's input goes up to the fleet:


```nix title="modules/inputs.nix" {14}
# Each feature declares the inputs it uses, next to the code that uses them.
{ sven, ... }:
{
  sven.aspects.core-inputs = {
    at = sven.kinds.fleet;
    inputs = {
      sven.url = "github:vic/sven";
      import-tree.url = "github:denful/import-tree";
      files.url = "github:sini/files";
    };
  };
  sven.aspects.web = {
    at = sven.host.web;
    inputs.nixpkgs.url = "https://nixos.org/channels/nixos-25.05/nixexprs.tar.xz"; # goes up to the fleet's `inputs` sink
    nixos = {
      services.nginx.enable = true;
      boot.loader.grub.device = "nodev";
      fileSystems."/" = {
        device = "/dev/sda1";
        fsType = "ext4";
      };
      nixpkgs.hostPlatform = "x86_64-linux";
      system.stateVersion = "25.05";
    };
  };
}
```

```nix title="modules/write.nix"
# flake.nix and README.md, generated from fleet data.
{ sven, lib, ... }:
let
  # one input, in nixfmt's layout: the generated file passes `nixfmt --check` unchanged
  input =
    n: v:
    "    ${n} = {\n"
    + lib.concatStrings (lib.mapAttrsToList (k: x: "      ${k} = ${builtins.toJSON x};\n") v)
    + "    };";
  hosts = lib.filter (s: s.kind == "host") sven.lib.graph.scopes;
in
{
  sven.aspects.write = {
    at = sven.kinds.fleet;
    files =
      { inputs, ... }:
      {
        files.file."flake.nix".text = ''
          # Generated from the `inputs` sink. Edit modules/, then run: nix run .#write-files
          {
            outputs =
              inputs:
              inputs.sven.lib.mkFlake {
                imports = [ (inputs.import-tree ./modules) ];
                _module.args = {
                  inherit inputs;
                  inherit (inputs.nixpkgs) lib;
                };
              };

            inputs = {
          ${lib.concatStringsSep "\n" (lib.mapAttrsToList input inputs)}
            };
          }
        '';
        files.file."README.md".text = ''
          # My fleet

          Generated from the fleet. Hosts: ${lib.concatMapStringsSep ", " (s: "`${s.name}`") hosts}.
        '';
      };
  };
}
```

The template's own `flake.nix` is that output. Remove the `web` aspect, and `nixpkgs` leaves the flake.


```nix title="flake.nix"
# Generated from the `inputs` sink. Edit modules/, then run: nix run .#write-files
{
  outputs =
    inputs:
    inputs.sven.lib.mkFlake {
      imports = [ (inputs.import-tree ./modules) ];
      _module.args = {
        inherit inputs;
        inherit (inputs.nixpkgs) lib;
      };
    };

  inputs = {
    files = {
      url = "github:sini/files";
    };
    import-tree = {
      url = "github:denful/import-tree";
    };
    nixpkgs = {
      url = "https://nixos.org/channels/nixos-25.05/nixexprs.tar.xz";
    };
    sven = {
      url = "github:vic/sven";
    };
  };
}
```

```sh frame="terminal"
nix run .#write-files   # write flake.nix and README.md
nix flake check         # fails when a committed file is stale
```

:::note[flake-file]
[vic/flake-file](https://github.com/vic/flake-file) generates `flake.nix` from a module option, with follows and checks built in.
The [reference fleet](https://github.com/vic/sven/blob/main/templates/fleet/modules/flake/flake-file.nix) uses it from a `flake-file` sink.
:::
